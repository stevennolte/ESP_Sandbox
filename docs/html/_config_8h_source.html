146 void begin() {
147 ssid = ConfigConstants::WiFi::DEFAULT_SSID;
148 password = ConfigConstants::WiFi::DEFAULT_PASSWORD;
149 last_check_time = 0;
150 force_ap_mode = ConfigConstants::WiFi::DEFAULT_AP_MODE;
151
152 // Initialize with default values from constants
153 max_attempts = ConfigConstants::WiFi::MAX_ATTEMPTS;
154 reconnect_attempts = ConfigConstants::WiFi::RECONNECT_ATTEMPTS;
155 retry_delay = ConfigConstants::WiFi::RETRY_DELAY;
156 }
157
165 void loadFromPreferences(Preferences& prefs) {
166 String saved_ssid = prefs.getString("wifi_ssid", "");
167 String saved_password = prefs.getString("wifi_password", "");
168
169 // Load AP mode preference
170 force_ap_mode = prefs.getBool("force_ap_mode", ConfigConstants::WiFi::DEFAULT_AP_MODE);
171
172 // Load WiFi parameters from preferences
173 max_attempts = prefs.getInt("wifi_max_attempts", ConfigConstants::WiFi::MAX_ATTEMPTS);
174 reconnect_attempts = prefs.getInt("wifi_reconnect_attempts", ConfigConstants::WiFi::RECONNECT_ATTEMPTS);
175 retry_delay = prefs.getInt("wifi_retry_delay", ConfigConstants::WiFi::RETRY_DELAY);
176
177 if (saved_ssid.length() > 0) {
178 ssid = saved_ssid;
179 password = saved_password;
180 Serial.printf("✓ WiFi credentials loaded: %s\n", ssid.c_str());
181 }
182
183 Serial.printf("✓ WiFi mode preference loaded: %s\n", force_ap_mode ? "Access Point" : "Station");
184 }
185
194 void saveCredentials(const String& newSSID, const String& newPassword) {
195 ssid = newSSID;
196 password = newPassword;
197 Preferences prefs;
198 prefs.begin("esp-config", false);
199 prefs.putString("wifi_ssid", ssid);
200 prefs.putString("wifi_password", password);
201 prefs.end();
202 Serial.printf("✓ WiFi credentials saved: %s\n", ssid.c_str());
203 }
204
213 void saveWiFiParams(int newMaxAttempts, int newReconnectAttempts, int newRetryDelay) {
214 max_attempts = newMaxAttempts;
215 reconnect_attempts = newReconnectAttempts;
216 retry_delay = newRetryDelay;
217
218 Preferences prefs;
219 prefs.begin("esp-config", false);
220 prefs.putInt("wifi_max_attempts", max_attempts);
221 prefs.putInt("wifi_reconnect_attempts", reconnect_attempts);
222 prefs.putInt("wifi_retry_delay", retry_delay);
223 prefs.end();
224
225 Serial.printf("✓ WiFi parameters saved: max_attempts=%d, reconnect_attempts=%d, retry_delay=%d\n",
226 max_attempts, reconnect_attempts, retry_delay);
227 }
