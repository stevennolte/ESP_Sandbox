71// --- Watchdog Functions ---
72void initWatchdog() {
73 Serial.println("Initializing software watchdog timer...");
74
75 // Check if this boot was due to a software watchdog timeout
76 preferences.begin("esp-config", true);
77 bool watchdogTimeout = preferences.getBool("watchdog_timeout", false);
78 preferences.end();
79
80 // Check if this is a recovery boot
81 esp_reset_reason_t resetReason = esp_reset_reason();
82 if (resetReason == ESP_RST_TASK_WDT || resetReason == ESP_RST_WDT || resetReason == ESP_RST_PANIC || watchdogTimeout) {
83 Serial.println("*** WATCHDOG/PANIC RESET DETECTED - ENTERING RECOVERY MODE ***");
84 recoveryMode = true;
85
86 // Store recovery mode flag in preferences
87 preferences.begin("esp-config", false);
88 preferences.putBool("recovery_mode", true);
89 preferences.putULong("recovery_time", millis());
90 preferences.putBool("watchdog_timeout", false); // Clear the flag
91 preferences.end();
92 } else {
93 // Clear recovery mode flag on normal boot
94 preferences.begin("esp-config", false);
95 preferences.putBool("recovery_mode", false);
96 preferences.putBool("watchdog_timeout", false);
97 preferences.end();
98 }
99
100 // Initialize software watchdog (we'll use a simple timer-based approach)
101 lastWatchdogFeed = millis();
102 Serial.printf("✓ Software watchdog initialized: %d second timeout\n", ConfigConstants::Timing::WATCHDOG_TIMEOUT);
103}

109void checkWatchdogTimeout() {
110 unsigned long timeSinceLastFeed = millis() - lastWatchdogFeed;
111 if (timeSinceLastFeed > (ConfigConstants::Timing::WATCHDOG_TIMEOUT * 1000)) {
112 Serial.println("*** SOFTWARE WATCHDOG TIMEOUT - RESTARTING ***");
113
114 // Mark as watchdog timeout for next boot
115 preferences.begin("esp-config", false);
116 preferences.putBool("watchdog_timeout", true);
117 preferences.end();
118
119 delay(1000);
120 ESP.restart();
121 }
122}

178// Function to load and process HTML template
179String loadHTMLTemplate(const char* filename) {
180 if (!LittleFS.exists(filename)) {
181 return "<!DOCTYPE html><html><body><h1>Error: Template file not found</h1></body></html>";
182 }
183
184 File file = LittleFS.open(filename, "r");
185 if (!file) {
186 return "<!DOCTYPE html><html><body><h1>Error: Could not open template file</h1></body></html>";
187 }
188
189 String html = file.readString();
190 file.close();
191
192 // Replace placeholders with actual values
193 html.replace("{{CLIENT_ID}}", config.client_id);
194 html.replace("{{LED_BRIGHTNESS}}", String(config.led_brightness));
195
196 // Use WiFiHelper for WiFi-related template variables
197 html = WiFiHelper::processWiFiTemplateVariables(html);
198
199 // Add template version info
200 preferences.begin("esp-config", true);
201 String templateCommit = preferences.getString("last_commit", "Unknown");
202 preferences.end();
203 html.replace("{{TEMPLATE_VERSION}}", templateCommit.length() > 7 ? templateCommit.substring(0, 7) : templateCommit);
204
205 return html;
206}

295void handleFileDownload() {
296 if (!server.hasArg("file")) {
297 server.send(400, "text/plain", "Missing file parameter");
298 return;
299 }
300
301 String filename = server.arg("file");
302 if (!filename.startsWith("/")) {
303 filename = "/" + filename;
304 }
305
306 if (!LittleFS.exists(filename)) {
307 server.send(404, "text/plain", "File not found");
308 return;
309 }
310
311 File file = LittleFS.open(filename, "r");
312 if (!file) {
313 server.send(500, "text/plain", "Failed to open file");
314 return;
315 }
316
317 server.streamFile(file, "application/octet-stream");
318 file.close();
319}

321void handleFileUpload() {
322 HTTPUpload& upload = server.upload();
323
324 if (upload.status == UPLOAD_FILE_START) {
325 String filename = "/" + upload.filename;
326 Serial.printf("Upload Start: %s\n", filename.c_str());
327
328 File file = LittleFS.open(filename, "w");
329 if (!file) {
330 Serial.println("Failed to create file");
331 return;
332 }
333 file.close();
334 } else if (upload.status == UPLOAD_FILE_WRITE) {
335 File file = LittleFS.open("/" + upload.filename, "a");
336 if (file) {
337 file.write(upload.buf, upload.currentSize);
338 file.close();
339 }
340 } else if (upload.status == UPLOAD_FILE_END) {
341 Serial.printf("Upload End: %s, Size: %u\n", upload.filename.c_str(), upload.totalSize);
342 }
343}

350// --- Firmware Upload Functions ---
351void handleFirmwareUpload() {
352 HTTPUpload& upload = server.upload();
353
354 if (upload.status == UPLOAD_FILE_START) {
355 Serial.printf("Firmware Upload Start: %s\n", upload.filename.c_str());
356 if (!Update.begin(UPDATE_SIZE_UNKNOWN)) {
357 Update.printError(Serial);
358 return;
359 }
360 } else if (upload.status == UPLOAD_FILE_WRITE) {
361 if (Update.write(upload.buf, upload.currentSize) != upload.currentSize) {
362 Update.printError(Serial);
363 }
364 } else if (upload.status == UPLOAD_FILE_END) {
365 if (Update.end(true)) {
366 Serial.printf("Firmware Update Success: %u bytes\n", upload.totalSize);
367 } else {
368 Update.printError(Serial);
369 }
370 }
371}

590bool downloadFileFromGitHub(const String& filePath, const String& localPath) {
591 HTTPClient http;
592 String url = "https://raw.githubusercontent.com/" + String(ConfigConstants::Firmware::GITHUB_REPO) + "/" + String(ConfigConstants::Firmware::GITHUB_BRANCH) + "/" + filePath;
593
594 http.begin(url);
595 http.addHeader("User-Agent", ConfigConstants::Network::USER_AGENT_TEMPLATE);
596 http.setTimeout(ConfigConstants::Network::HTTP_TIMEOUT_LONG);
597
598 int httpCode = http.GET();
599 bool success = false;
600
601 if (httpCode == HTTP_CODE_OK) {
602 String payload = http.getString();
603
604 if (payload.length() > 0) {
605 File file = LittleFS.open(localPath, "w");
606 if (file) {
607 size_t written = file.print(payload);
608 file.close();
609 success = written > 0;
610 Serial.printf("Downloaded %d bytes to %s\n", written, localPath.c_str());
611 }
612 }
613 } else {
614 Serial.printf("Download failed: HTTP %d\n", httpCode);
615 }
616
617 http.end();
618 return success;
619}

1054void loop() {
1055 unsigned long currentTime = millis();
1056
1057 // Update status indicator (must be called for animations)
1058 statusLED.update();
1059
1060 // Feed the watchdog timer at the start of each loop
1061 if (!recoveryMode) {
1062 feedWatchdog();
1063
1064 // Check for watchdog timeout (separate check for safety)
1065 checkWatchdogTimeout();
1066 }
1067
1068 // Handle recovery mode separately
1069 if (recoveryMode) {
1070 server.handleClient();
1071 delay(100); // Shorter delay in recovery mode
1072 return;
1073 }
1074
1075 // Handle web server requests
1076 server.handleClient();
1077
1078 // Periodic tasks with timing
1079
1080 // WiFi connection monitoring (every 30 seconds)
1081 if (config.shouldCheckWiFi(currentTime)) {
1082 WiFiHelper::checkConnection();
1083 config.updateWiFiCheckTime(currentTime);
1084
1085 // Update status based on WiFi connection
1086 if (!WiFiHelper::isConnected()) {
1087 statusLED.showError();
1088 } else {
1089 statusLED.showNormal();
1090 }
1091 }
1092
1093 // OTA update checking (every 5 minutes)
1094 if (config.shouldCheckUpdates(currentTime)) {
1095 statusLED.showInfo(); // Show checking for updates
1096 otaUpdater.checkForUpdates();
1097 config.updateUpdateCheckTime(currentTime);
1098 statusLED.showNormal(); // Back to normal
1099 }
1100
1101 delay(ConfigConstants::Timing::MAIN_LOOP_DELAY);
1102}
