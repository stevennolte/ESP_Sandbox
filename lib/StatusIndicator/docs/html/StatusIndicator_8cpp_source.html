217void StatusIndicator::update() {
218 if (!enabled || currentMode == IndicatorMode::OFF) {
219 return;
220 }
221
222 unsigned long currentTime = millis();
223
224 switch (currentMode) {
225 case IndicatorMode::SOLID:
226 // Set solid color based on status
227 if (ledType == LEDType::RGB_LED) {
228 uint8_t r, g, b;
229 getStatusColor(currentStatus, r, g, b);
230 setLEDColor(r, g, b);
231 } else {
232 setSingleLED(brightness);
233 }
234 break;
235
236 case IndicatorMode::PULSE:
237 if (currentTime - lastUpdate >= pulseInterval) {
238 updatePulse();
239 lastUpdate = currentTime;
240 }
241 break;
242
243 case IndicatorMode::BLINK_SLOW:
244 if (currentTime - lastUpdate >= 500) { // 500ms intervals
245 updateBlink();
246 lastUpdate = currentTime;
247 }
248 break;
249
250 case IndicatorMode::BLINK_FAST:
251 if (currentTime - lastUpdate >= 150) { // 150ms intervals
252 updateBlink();
253 lastUpdate = currentTime;
254 }
255 break;
256
257 case IndicatorMode::BREATHE:
258 if (currentTime - lastUpdate >= 25) { // 25ms intervals for smooth breathing
259 updateBreathe();
260 lastUpdate = currentTime;
261 }
262 break;
263
264 case IndicatorMode::RGB_CYCLE:
265 if (ledType == LEDType::RGB_LED && currentTime - lastUpdate >= 50) {
266 updateRGBCycle();
267 lastUpdate = currentTime;
268 }
269 break;
270
271 default:
272 break;
273 }
274}

312void StatusIndicator::updateBreathe() {
313 // Smooth breathing effect
314 if (pulseDirection) {
315 pulseValue += 2;
316 if (pulseValue >= brightness) {
317 pulseValue = brightness;
318 pulseDirection = false;
319 }
320 } else {
321 pulseValue -= 2;
322 if (pulseValue <= 0) {
323 pulseValue = 0;
324 pulseDirection = true;
325 }
326 }
327
328 if (ledType == LEDType::RGB_LED) {
329 uint8_t r, g, b;
330 getStatusColor(currentStatus, r, g, b);
331 // Scale colors by pulse value
332 r = (r * pulseValue) / MAX_BRIGHTNESS;
333 g = (g * pulseValue) / MAX_BRIGHTNESS;
334 b = (b * pulseValue) / MAX_BRIGHTNESS;
335 setLEDColor(r, g, b);
336 } else {
337 setSingleLED(pulseValue);
338 }
339}

465void StatusIndicator::hsvToRgb(uint16_t hue, uint8_t sat, uint8_t val, uint8_t& red, uint8_t& green, uint8_t& blue) {
466 uint16_t h = hue % 360;
467 uint8_t s = sat;
468 uint8_t v = val;
469
470 uint8_t c = (v * s) / 255;
471 uint16_t x = (c * (60 - abs((h % 120) - 60))) / 60;
472 uint8_t m = v - c;
473
474 if (h < 60) {
475 red = c + m;
476 green = x + m;
477 blue = m;
478 } else if (h < 120) {
479 red = x + m;
480 green = c + m;
481 blue = m;
482 } else if (h < 180) {
483 red = m;
484 green = c + m;
485 blue = x + m;
486 } else if (h < 240) {
487 red = m;
488 green = x + m;
489 blue = c + m;
490 } else if (h < 300) {
491 red = x + m;
492 green = m;
493 blue = c + m;
494 } else {
495 red = c + m;
496 green = m;
497 blue = x + m;
498 }
