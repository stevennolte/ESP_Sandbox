221void StatusIndicator::update() {
222 if (!enabled || currentMode == IndicatorMode::OFF) {
223 return;
224 }
225
226 unsigned long currentTime = millis();
227
228 switch (currentMode) {
229 case IndicatorMode::SOLID:
230 // Set solid color based on status
231 if (ledType == LEDType::RGB_LED) {
232 uint8_t r, g, b;
233 getStatusColor(currentStatus, r, g, b);
234 setLEDColor(r, g, b);
235 } else {
236 setSingleLED(brightness);
237 }
238 break;
239
240 case IndicatorMode::PULSE:
241 if (currentTime - lastUpdate >= pulseInterval) {
242 updatePulse();
243 lastUpdate = currentTime;
244 }
245 break;
246
247 case IndicatorMode::BLINK_SLOW:
248 if (currentTime - lastUpdate >= 500) { // 500ms intervals
249 updateBlink();
250 lastUpdate = currentTime;
251 }
252 break;
253
254 case IndicatorMode::BLINK_FAST:
255 if (currentTime - lastUpdate >= 150) { // 150ms intervals
256 updateBlink();
257 lastUpdate = currentTime;
258 }
259 break;
260
261 case IndicatorMode::BREATHE:
262 if (currentTime - lastUpdate >= 25) { // 25ms intervals for smooth breathing
263 updateBreathe();
264 lastUpdate = currentTime;
265 }
266 break;
267
268 case IndicatorMode::RGB_CYCLE:
269 if (ledType == LEDType::RGB_LED && currentTime - lastUpdate >= 50) {
270 updateRGBCycle();
271 lastUpdate = currentTime;
272 }
273 break;
274
275 default:
276 break;
277 }
278}

316void StatusIndicator::updateBreathe() {
317 // Smooth breathing effect
318 if (pulseDirection) {
319 pulseValue += 2;
320 if (pulseValue >= brightness) {
321 pulseValue = brightness;
322 pulseDirection = false;
323 }
324 } else {
325 pulseValue -= 2;
326 if (pulseValue <= 0) {
327 pulseValue = 0;
328 pulseDirection = true;
329 }
330 }
331
332 if (ledType == LEDType::RGB_LED) {
333 uint8_t r, g, b;
334 getStatusColor(currentStatus, r, g, b);
335 // Scale colors by pulse value
336 r = (r * pulseValue) / MAX_BRIGHTNESS;
337 g = (g * pulseValue) / MAX_BRIGHTNESS;
338 b = (b * pulseValue) / MAX_BRIGHTNESS;
339 setLEDColor(r, g, b);
340 } else {
341 setSingleLED(pulseValue);
342 }
343}

489void StatusIndicator::hsvToRgb(uint16_t hue, uint8_t sat, uint8_t val, uint8_t& red, uint8_t& green, uint8_t& blue) {
490 uint16_t h = hue % 360;
491 uint8_t s = sat;
492 uint8_t v = val;
493
494 uint8_t c = (v * s) / 255;
495 uint16_t x = (c * (60 - abs((h % 120) - 60))) / 60;
496 uint8_t m = v - c;
497
498 if (h < 60) {
499 red = c + m;
500 green = x + m;
501 blue = m;
502 } else if (h < 120) {
503 red = x + m;
504 green = c + m;
505 blue = m;
506 } else if (h < 180) {
507 red = m;
508 green = c + m;
509 blue = x + m;
510 } else if (h < 240) {
511 red = m;
512 green = x + m;
513 blue = c + m;
514 } else if (h < 300) {
515 red = x + m;
516 green = m;
517 blue = c + m;
518 } else {
519 red = c + m;
520 green = m;
521 blue = x + m;
522 }

532LEDType StatusIndicator::detectBoardLEDType() {
533 String boardType = getBoardTypeName();
534
535 // Check for ESP32-S3-DevKitC-1 specifically (has WS2812 RGB LED on GPIO38)
536 if (boardType.indexOf("ESP32-S3") >= 0 &&
537 (boardType.indexOf("DevKit") >= 0 || boardType.indexOf("DEVKIT") >= 0)) {
538 return LEDType::WS2812_LED;
